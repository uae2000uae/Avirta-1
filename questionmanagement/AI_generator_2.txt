```python
def search_wikipedia_pages(topic: str, lang: str = "en", limit: int = 10) -> list:
    url = f"https://{lang}.wikipedia.org/w/api.php"
    params = {
        "action": "query",
        "list": "search",
        "srsearch": topic,
        "format": "json",
        "srlimit": limit

    response = requests.get(
        url,
```

```python
def is_duplicate_question(new_q: dict, existing_questions: list) -> bool:
    new_text = _normalize_question_text(new_q.get("question", ""))
    new_answer = _normalize_question_text(new_q.get("correct_answer", ""))
    new_fact_id = new_q.get("fact_id")
    new_source_fact = _normalize_question_text(new_q.get("source_fact_used", ""))

    for old_q in existing_questions:
        old_text = _normalize_question_text(old_q.get("question", ""))
        old_answer = _normalize_question_text(old_q.get("correct_answer", ""))
        old_fact_id = old_q.get("fact_id")
        old_source_fact = _normalize_question_text(old_q.get("source_fact_used", ""))

        if new_fact_id and old_fact_id and new_fact_id == old_fact_id:
            return True

        if new_source_fact and old_source_fact and new_source_fact == old_source_fact:
            return True

        if new_text and old_text and new_text == old_text:
            return True

        if new_answer and old_answer and new_answer == old_answer and new_source_fact == old_source_fact:
            return True

    return False
```

Then filter before saving:

```python
def filter_duplicates(generated_questions, existing_questions):
    approved = []
    rejected = []

    memory = list(existing_questions)

    for q in generated_questions:
        if is_duplicate_question(q, memory):
            rejected.append(q)
            continue

        approved.append(q)
        memory.append(q)

    return approved, rejected
```

```python
def generate_until_target_count(
    client,
    model,
    topic,
    category,
    facts,
    existing_questions,
    target_count,
    distribution,
    max_rounds=4
):
    final_questions = []
    available_facts = list(facts)

    for round_no in range(max_rounds):
        missing = target_count - len(final_questions)

        if missing <= 0:
            break

        used_fact_ids = {
            q.get("fact_id")
            for q in existing_questions + final_questions
            if q.get("fact_id")
        }

        available_facts = [
            f for f in available_facts
            if f.get("fact_id") not in used_fact_ids
        ]

        if not available_facts:
            break

        generated = call_ai_generate_questions(
            client=client,
            model=model,
            topic=topic,
            category=category,
            facts=available_facts,
            question_count=missing,
            distribution=distribution
        )

        validated = call_ai_validate_questions(
            client=client,
            model=model,
            facts=available_facts,
            questions=generated
        )

        approved = extract_approved_questions(validated)

        unique, rejected = filter_duplicates(
            approved,
            existing_questions + final_questions
        )

        final_questions.extend(unique)

    return final_questions
```

This loop is one of the most important missing pieces.

## 8) Add hard validation for multiple choice

AI validation may say a question is fine even when the options are broken. Add code validation:

```python
def code_validate_question(q: dict) -> tuple[bool, list]:
    issues = []

    qtype = _normalize_question_type(q.get("type"))
    q["type"] = qtype

    if not q.get("question"):
        issues.append("missing_question")

    if q.get("points") not in [100, 200, 300, 400, 500]:
        issues.append("invalid_points")

    if not q.get("correct_answer"):
        issues.append("missing_correct_answer")

    if qtype == "multiple_choice":
        options = q.get("options") or []

        if len(options) != 4:
            issues.append("multiple_choice_must_have_4_options")

        normalized_options = [_normalize_question_text(o) for o in options]
        normalized_answer = _normalize_question_text(q.get("correct_answer", ""))

        if normalized_answer not in normalized_options:
            issues.append("correct_answer_not_in_options")

        if len(set(normalized_options)) != len(normalized_options):
            issues.append("duplicate_options")

    if qtype == "true_false":
        valid_answers = {"true", "false"}
        if _normalize_question_text(q.get("correct_answer", "")) not in valid_answers:
            issues.append("true_false_answer_must_be_true_or_false")

    if not q.get("fact_id"):
        issues.append("missing_fact_id")

    if not q.get("source_url"):
        issues.append("missing_source_url")

    return len(issues) == 0, issues
```

Then:

```python
def code_filter_valid_questions(questions):
    valid = []
    rejected = []

    for q in questions:
        ok, issues = code_validate_question(q)
        if ok:
            valid.append(q)
        else:
            rejected.append({
                "question": q,
                "issues": issues
            })

    return valid, rejected
```
